cmake_minimum_required(VERSION 3.24)
project(CudaSearchLab VERSION 1.0.0 LANGUAGES CXX)
option(CUDASEARCH_ENABLE_CUDA "Build the NVIDIA CUDA backend" ON)
option(CUDASEARCH_SANITIZERS "Enable AddressSanitizer and UBSan in CPU builds" OFF)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(cudasearch src/graph.cpp src/cpu.cpp)
target_include_directories(cudasearch PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_compile_options(cudasearch PRIVATE $<$<COMPILE_LANGUAGE:CXX>:-Wall;-Wextra;-Wpedantic>)
if(CUDASEARCH_ENABLE_CUDA)
    if(NOT DEFINED CMAKE_CUDA_ARCHITECTURES)
        set(CMAKE_CUDA_ARCHITECTURES 75 CACHE STRING "CUDA architectures")
    endif()
    enable_language(CUDA)
    find_package(CUDAToolkit REQUIRED)
    target_sources(cudasearch PRIVATE src/cuda_search.cu)
    target_link_libraries(cudasearch PUBLIC CUDA::cudart)
    set_target_properties(cudasearch PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED YES)
    target_compile_options(cudasearch PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:-lineinfo>)
else()
    target_sources(cudasearch PRIVATE src/cuda_stub.cpp)
endif()
if(CUDASEARCH_SANITIZERS)
    if(CUDASEARCH_ENABLE_CUDA)
        message(FATAL_ERROR "Use CUDASEARCH_ENABLE_CUDA=OFF for the CPU sanitizer build")
    endif()
    target_compile_options(cudasearch PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(cudasearch PUBLIC -fsanitize=address,undefined)
endif()
add_executable(searchlab src/main.cpp)
target_link_libraries(searchlab PRIVATE cudasearch)
add_executable(search_tests tests/search_tests.cpp)
target_link_libraries(search_tests PRIVATE cudasearch)
enable_testing()
add_test(NAME cpu_correctness COMMAND search_tests)
add_test(NAME cli_smoke COMMAND searchlab --backend cpu --algo astar --grid ${CMAKE_CURRENT_SOURCE_DIR}/examples/maze.txt --validate)
add_test(NAME cli_json_csv COMMAND searchlab --backend cpu --algo all --graph ${CMAKE_CURRENT_SOURCE_DIR}/examples/weighted.edgelist --depth 10 --validate --json cli_results.json --csv cli_results.csv)
if(CUDASEARCH_ENABLE_CUDA)
    add_test(NAME cuda_correctness COMMAND search_tests --cuda)
    set_tests_properties(cuda_correctness PROPERTIES SKIP_RETURN_CODE 77)
endif()
